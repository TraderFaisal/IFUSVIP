/* ══════════════════════════════════════════════
   ✅ 2. TRADING ANIMATED BACKGROUND
   — Scrolling candlestick chart + price line
══════════════════════════════════════════════ */
(function() {
    const canvas = document.getElementById('tradingBg');
    const ctx    = canvas.getContext('2d');

    function resize() {
        canvas.width  = window.innerWidth;
        canvas.height = window.innerHeight;
    }
    resize();
    window.addEventListener('resize', resize);

    // Generate price data (random walk)
    const totalCandles = 80;
    const candles = [];
    let price = 0.5;

    for (let i = 0; i < totalCandles; i++) {
        const open  = price;
        const move  = (Math.random() - 0.48) * 0.06;
        price = Math.max(0.1, Math.min(0.9, price + move));
        const close = price;
        const high  = Math.max(open, close) + Math.random() * 0.025;
        const low   = Math.min(open, close) - Math.random() * 0.025;
        candles.push({ open, close, high, low });
    }

    // Scrolling price line points
    const linePoints = [];
    for (let i = 0; i < totalCandles; i++) {
        linePoints.push((candles[i].open + candles[i].close) / 2);
    }

    let scrollX = 0;
    let animFrame = 0;

    function drawBg() {
        const W = canvas.width;
        const H = canvas.height;

        // Dark base
        ctx.fillStyle = '#06090e';
        ctx.fillRect(0, 0, W, H);

        // Soft radial glows
        const g1 = ctx.createRadialGradient(W*0.15, H*0.15, 0, W*0.15, H*0.15, W*0.4);
        g1.addColorStop(0, 'rgba(160,60,255,0.09)');
        g1.addColorStop(1, 'transparent');
        ctx.fillStyle = g1;
        ctx.fillRect(0, 0, W, H);

        const g2 = ctx.createRadialGradient(W*0.85, H*0.85, 0, W*0.85, H*0.85, W*0.4);
        g2.addColorStop(0, 'rgba(0,210,255,0.08)');
        g2.addColorStop(1, 'transparent');
        ctx.fillStyle = g2;
        ctx.fillRect(0, 0, W, H);

        // Grid lines
        ctx.strokeStyle = 'rgba(255,255,255,0.03)';
        ctx.lineWidth = 1;
        const gridStep = 50;
        for (let x = 0; x < W; x += gridStep) {
            ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
        }
        for (let y = 0; y < H; y += gridStep) {
            ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
        }

        // Candle settings
        const candleW  = 14;
        const candleGap = 8;
        const totalW   = (candleW + candleGap) * totalCandles;
        const chartTop    = H * 0.15;
        const chartBottom = H * 0.80;
        const chartH      = chartBottom - chartTop;

        const offsetX = scrollX % totalW;

        // Draw candles (draw twice for seamless scroll)
        for (let rep = -1; rep <= 1; rep++) {
            for (let i = 0; i < totalCandles; i++) {
                const c = candles[i];
                const x = rep * totalW + i * (candleW + candleGap) - offsetX + W * 0.05;
                if (x + candleW < 0 || x > W) continue;

                const oY = chartBottom - c.open  * chartH;
                const cY = chartBottom - c.close * chartH;
                const hY = chartBottom - c.high  * chartH;
                const lY = chartBottom - c.low   * chartH;

                const bull = c.close >= c.open;
                const color     = bull ? 'rgba(0,220,130,0.7)' : 'rgba(255,70,90,0.7)';
                const colorDim  = bull ? 'rgba(0,220,130,0.2)' : 'rgba(255,70,90,0.2)';

                // Wick
                ctx.strokeStyle = color;
                ctx.lineWidth   = 1.5;
                ctx.beginPath();
                ctx.moveTo(x + candleW/2, hY);
                ctx.lineTo(x + candleW/2, lY);
                ctx.stroke();

                // Body
                const bodyTop = Math.min(oY, cY);
                const bodyH   = Math.max(Math.abs(cY - oY), 2);
                ctx.fillStyle = color;
                ctx.fillRect(x, bodyTop, candleW, bodyH);

                // Glow behind body
                ctx.fillStyle = colorDim;
                ctx.fillRect(x - 2, bodyTop - 2, candleW + 4, bodyH + 4);
            }
        }

        // Smooth price line (green trend line)
        ctx.beginPath();
        ctx.strokeStyle = 'rgba(0,255,180,0.55)';
        ctx.lineWidth   = 2;
        ctx.shadowColor = 'rgba(0,255,180,0.4)';
        ctx.shadowBlur  = 10;

        for (let rep = -1; rep <= 1; rep++) {
            for (let i = 0; i < linePoints.length; i++) {
                const x = rep * totalW + i * (candleW + candleGap) + candleW/2 - offsetX + W * 0.05;
                const y = chartBottom - linePoints[i] * chartH;
                if (i === 0 && rep === -1) ctx.moveTo(x, y);
                else ctx.lineTo(x, y);
            }
        }
        ctx.stroke();
        ctx.shadowBlur = 0;

        // Floating % labels
        const labels = ['+2.4%', '+0.8%', '-1.2%', '+3.1%', '+0.5%'];
        ctx.font = 'bold 12px Poppins, sans-serif';
        labels.forEach((lbl, i) => {
            const lx = (W / labels.length) * i + W / (labels.length * 2);
            const ly = H * 0.88 + Math.sin(animFrame * 0.02 + i) * 6;
            ctx.fillStyle = lbl.startsWith('+') ? 'rgba(0,220,130,0.5)' : 'rgba(255,70,90,0.5)';
            ctx.fillText(lbl, lx, ly);
        });

        // Scroll
        scrollX += 0.5;
        animFrame++;
        requestAnimationFrame(drawBg);
    }
    drawBg();
})();

/* ══════════════════════════════════════════════
   ✅ 1. ELECTRIC LIGHTNING BORDER
══════════════════════════════════════════════ */
(function() {
    const svg   = document.getElementById('lightningCanvas');
    const flash = document.getElementById('flashOverlay');

    // Fractal lightning bolt generator
    function buildBolt(x1, y1, x2, y2, roughness, depth) {
        if (depth === 0) return [[x1,y1],[x2,y2]];
        const mx = (x1+x2)/2 + (Math.random()-0.5)*roughness;
        const my = (y1+y2)/2 + (Math.random()-0.5)*roughness;
        const L  = buildBolt(x1,y1,mx,my,roughness*0.58,depth-1);
        const R  = buildBolt(mx,my,x2,y2,roughness*0.58,depth-1);
        return [...L.slice(0,-1),...R];
    }

    function pts(points) {
        return points.map(p => p[0].toFixed(1)+','+p[1].toFixed(1)).join(' ');
    }

    function addPolyline(points, stroke, width, opacity, blur) {
        const pl = document.createElementNS('http://www.w3.org/2000/svg','polyline');
        pl.setAttribute('points', pts(points));
        pl.setAttribute('stroke', stroke);
        pl.setAttribute('stroke-width', String(width));
        pl.setAttribute('fill', 'none');
        pl.setAttribute('stroke-linecap', 'round');
        pl.setAttribute('stroke-linejoin', 'round');
        pl.setAttribute('opacity', String(opacity));
        if (blur) pl.style.filter = `blur(${blur}px)`;
        svg.appendChild(pl);
        return pl;
    }

    function fadeOut(elements, duration) {
        let op    = 1;
        const step = 1 / (duration / 30);
        const timer = setInterval(() => {
            op -= step;
            if (op <= 0) {
                clearInterval(timer);
                elements.forEach(el => { try { svg.removeChild(el); } catch(e){} });
            } else {
                elements.forEach(el => el.setAttribute('opacity', Math.max(0,parseFloat(el.getAttribute('opacity'))-step).toFixed(2)));
            }
        }, 30);
    }

    function getSize() {
        const el = document.getElementById('lightningCanvas');
        return { w: el.clientWidth, h: el.clientHeight };
    }

    // Short border strike
    function drawStrike() {
        const { w, h } = getSize();
        const pad = 20;
        const edge = Math.floor(Math.random()*4);
        let x1,y1,x2,y2;

        switch(edge){
            case 0: x1=pad+Math.random()*(w-2*pad); y1=pad;   x2=x1+(Math.random()-0.5)*100; y2=pad;   break;
            case 1: x1=w-pad; y1=pad+Math.random()*(h-2*pad); x2=w-pad; y2=y1+(Math.random()-0.5)*100; break;
            case 2: x1=pad+Math.random()*(w-2*pad); y1=h-pad; x2=x1+(Math.random()-0.5)*100; y2=h-pad; break;
            case 3: x1=pad; y1=pad+Math.random()*(h-2*pad);   x2=pad;   y2=y1+(Math.random()-0.5)*100; break;
        }

        const bolt = buildBolt(x1,y1,x2,y2,22,6);
        const col  = ['#ffffff','#c8d8ff','#aaddff','#ffffcc'][Math.floor(Math.random()*4)];

        const els = [
            addPolyline(bolt, col,    8, 0.18, 5),
            addPolyline(bolt, col,  2.5, 0.65, 0),
            addPolyline(bolt,'#fff',  1, 1.0,  0),
        ];

        flash.style.opacity = '1';
        setTimeout(()=>{ flash.style.opacity='0'; }, 55);
        fadeOut(els, 350);

        // Double strike
        if (Math.random() < 0.38) setTimeout(drawStrike, 70+Math.random()*80);
    }

    // Long bolt across card edge to edge
    function drawLongBolt() {
        const { w, h } = getSize();
        const pad = 20;
        function edgePt(e){
            switch(e){
                case 0: return [pad+Math.random()*(w-2*pad), pad];
                case 1: return [w-pad, pad+Math.random()*(h-2*pad)];
                case 2: return [pad+Math.random()*(w-2*pad), h-pad];
                case 3: return [pad, pad+Math.random()*(h-2*pad)];
            }
        }
        const e1 = Math.floor(Math.random()*4);
        const e2 = (e1+1+Math.floor(Math.random()*2))%4;
        const [x1,y1] = edgePt(e1);
        const [x2,y2] = edgePt(e2);
        const bolt = buildBolt(x1,y1,x2,y2,45,7);
        const col  = Math.random()<0.5 ? '#c8d8ff' : '#ffe066';

        const els = [
            addPolyline(bolt, col,  12, 0.15, 7),
            addPolyline(bolt, col, 2.5, 0.70, 0),
            addPolyline(bolt,'#fff', 1.2, 0.9, 0),
        ];

        flash.style.opacity = '0.9';
        setTimeout(()=>{ flash.style.opacity='0'; }, 75);
        fadeOut(els, 400);
    }

    function scheduleBolt() {
        drawStrike();
        setTimeout(scheduleBolt, 450 + Math.random()*2200);
    }
    function scheduleLong() {
        drawLongBolt();
        setTimeout(scheduleLong, 3200 + Math.random()*4500);
    }

    setTimeout(scheduleBolt, 700);
    setTimeout(scheduleLong, 2000);
})();
